func loadRecords() ([]Record, error) {
	data, err := ioutil.ReadFile("records.json")
	if os.IsNotExist(err) {
		return []Record{}, nil
	}
	if err != nil {
		return nil, err
	}
	var records []Record
	err = json.Unmarshal(data, &records)
	return records, err
}
